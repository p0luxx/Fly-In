class Connection:

    def id(self) -> str:
        return f"{self.from_zone}-{self.to_zone}"

        if current_zone == self.from_zone:

        raise ValueError(
            f"Zone: {current_zone} is not part of connection {self.id}"
        )

class Graph:
    def __init__(self) -> None:
        self.start: Hub | None = None
        self.end: Hub | None = None
        self.n_drones: int = 0
        self.link_dict: dict[str, Connection] = {}
        self.nodes_dict: dict[str, Hub] = {}
        self.adj: dict[str, list[Connection]] = {}

    def test(self):
        print("Link dict:")
        for item in self.link_dict.items():
            print(item)
        print("Nodes dict")
        for item in self.nodes_dict.items():
            print(item)

    def add_node(self, node: Hub) -> None:
        if node.name in self.nodes_dict:
            raise ValueError(f"The node '{node.name}' is duplicated in the map.")
        if node.directive == "start_hub":
            self.start = node
        elif node.directive == "end_hub":
            self.end = node
        self.nodes_dict[node.name] = node
        self.adj[node.name] = []

    def add_link(self, link: Connection) -> None:
        if link.from_zone in self.nodes_dict and link.to_zone in self.nodes_dict:
            if link.from_zone != link.to_zone:
                self.link_dict[link.id] = link
                self.adj[link.from_zone].append(link)
                self.adj[link.to_zone].append(link)
            else:
                raise ValueError(
                    "A node connected to itself in a bidirectional manner "
                    "has been detected.\nClosing Fly-In..."
                )
        else:
            raise ValueError(
                "An orphaned node has been detected.\nClosing Fly-In..."
            )

    def get_neighbors(self, zone_name: str) -> list[Connection]:
        return self.adj.get(zone_name, [])

"""Load and validate map files for the Fly-In application."""
from __future__ import annotations

from typing import Any

from pydantic import BaseModel, Field, model_validator

from Backend.Zones.conection import Connection
from Backend.Zones.hub import Hub
from Backend.Zones.hub_type import HubType
import sys


class MapParser(BaseModel):
    file_name: str
    content: list[str] = Field(default_factory=list, exclude=True)
    nb_drones: int = 0
    hubs: list[Hub] = Field(default_factory=list)
    connections: list[Connection] = Field(default_factory=list)

    @model_validator(mode="after")
    def process_file_automatically(self) -> MapParser:
        self._read_file()
        self._parse_and_route()
        start_count = sum(hub.directive == "start_hub" for hub in self.hubs)
        end_count = sum(hub.directive == "end_hub" for hub in self.hubs)
        if start_count != 1 or end_count != 1:
            raise ValueError(
                "The map must contain exactly one start_hub and one end_hub."
            )
        return self

    def _read_file(self) -> None:
        try:
            with open(self.file_name, "r", encoding="utf-8") as map_file:
                raw_lines = map_file.readlines()
        except PermissionError as error:
            raise PermissionError(
                f"Permission denied while reading map file: {self.file_name}"
            ) from error
        except FileNotFoundError as error:
            raise FileNotFoundError(
                f"Map file not found: {self.file_name}"
            ) from error
        except IsADirectoryError as error:
            raise IsADirectoryError(
                f"Expected a map file, but a directory was provided: {error}"
            ) from error

        self.content = [
            line.strip()
            for line in raw_lines
            if line.strip() and not line.strip().startswith("#")
        ]

    def _parse_and_route(self) -> None:
        try:
            for line in self.content:
                if ":" not in line:
                    raise ValueError("Invalid map syntax: missing ':' separator.")
                key, value = (part.strip() for part in line.split(":", 1))
                match key:
                    case "nb_drones":
                        self._drone_processing(value)
                    case "start_hub" | "hub" | "end_hub":
                        self._hub_processor(key, value)
                    case "connection":
                        self._connection_processor(value)
                    case _:
                        raise ValueError(f"Unknown map directive: {key}")
        except ValueError as error:
            raise ValueError(f"Invalid map: {error}") from error

    def _drone_processing(self, value: str) -> None:
        self.nb_drones = int(value)

    def _hub_processor(self, directive: str, value: str) -> None:
        parts = value.split("[", 1)
        coordinates = parts[0].split()
        if len(coordinates) != 3:
            raise ValueError("A hub must define a name and x/y coordinates.")
        name, x_str, y_str = coordinates
        if "-" in name:
            print(f"The node name cannot contain hyphens: '{name}'")
            sys.exit()
        extras: dict[str, Any] = {}
        if len(parts) > 1:
            extra_raw = parts[1].strip()
            if not extra_raw.endswith("]"):
                raise ValueError("Invalid hub syntax: missing closing ']'.")
            for item in extra_raw[:-1].split():
                key, value_part = item.split("=", 1)
                try:
                    if value_part.lstrip("-").isnumeric():
                        extras[key] = int(value_part)
                    elif key == "zone":
                        extras[key] = HubType[value_part.upper()]
                    else:
                        extras[key] = value_part
                except KeyError as error:
                    raise ValueError(
                        "Invalid hub zone type; use a supported HubType."
                    ) from error
        try:
            self.hubs.append(
                Hub(
                    directive=directive,
                    name=name,
                    x=int(x_str),
                    y=int(y_str),
                    **extras,
                )
            )
        except TypeError as e:
            raise TypeError(
                "Unknown parameter within [...]"
            ) from e

    def _connection_processor(self, value: str) -> None:
        parts = value.strip().split("[", 1)
        route_str = parts[0].strip()
        if "-" not in route_str:
            raise ValueError(
                f"Invalid connection syntax (missing '-'): '{route_str}'"
            )

        from_zone, to_zone = (part.strip() for part in route_str.split("-", 1))
        if not from_zone or not to_zone:
            raise ValueError(f"Connection endpoints cannot be empty: '{route_str}'")

        capacity = 1
        if len(parts) > 1:
            extra_raw = parts[1].strip()
            if not extra_raw.endswith("]"):
                raise ValueError("Invalid connection syntax: missing closing ']'.")
            for item in extra_raw[:-1].split():
                if "=" not in item:
                    continue
                key, raw_value = item.split("=", 1)
                if key == "max_link_capacity":
                    if not raw_value.isdigit() or int(raw_value) <= 0:
                        raise ValueError(
                            "max_link_capacity must be a positive integer."
                        )
                    capacity = int(raw_value)

        self.connections.append(
            Connection(
                from_zone=from_zone,
                to_zone=to_zone,
                max_link_capacity=capacity,
            )
        )

"""Graph validation, capacity reservations, and space-time route search."""
from __future__ import annotations

import heapq

from Backend.Solver.models import ScheduledStep, _SearchNode
from Backend.Zones.conection import Connection
from Backend.Zones.hub import Hub
from Backend.Zones.hub_type import HubType
from Backend.Zones.graph import Graph

__all__ = ["_Pathfinding"]


class _Pathfinding:
    """Route-search behavior mixed into the public solver."""

    graph: Graph
    heuristics: dict[str, int]
    hub_reservations: dict[str, dict[int, int]]
    link_reservations: dict[str, dict[int, int]]
    max_turns_limit: int

    def _validate_graph(self) -> tuple[Hub, Hub]:
        if self.graph.start is None or self.graph.end is None:
            raise ValueError(
                "The graph must define both a 'start_hub' and an 'end_hub'."
            )
        if self.graph.n_drones < 0:
            raise ValueError("The number of drones cannot be negative.")
        if (
            self.graph.start.zone == HubType.BLOCKED
            or self.graph.end.zone == HubType.BLOCKED
        ):
            raise ValueError("The start or end hub is blocked (BLOCKED).")
        return self.graph.start, self.graph.end

    @staticmethod
    def _canonical_link_id(link: Connection) -> str:
        """Return a direction-independent identifier for a connection."""
        a, b = sorted((link.from_zone, link.to_zone))
        return f"{a}<->{b}"

    @staticmethod
    def _orient_connection(
        link: Connection, from_zone: str, to_zone: str
    ) -> Connection:
        """Orient a connection in the direction of travel."""
        if link.from_zone == from_zone and link.to_zone == to_zone:
            return link
        return Connection(
            from_zone=from_zone,
            to_zone=to_zone,
            max_link_capacity=link.max_link_capacity,
        )

    def _compute_backward_heuristics(self) -> None:
        """Compute the admissible minimum-turn distance to the end hub."""
        start_hub, end_hub = self._validate_graph()
        dist: dict[str, int] = {end_hub.name: 0}
        pq: list[tuple[int, str]] = [(0, end_hub.name)]

        while pq:
            current_cost, current_name = heapq.heappop(pq)
            if current_cost > dist.get(current_name, float("inf")):
                continue

            current_hub = self.graph.nodes_dict[current_name]
            step_cost = 2 if current_hub.zone == HubType.RESTRICTED else 1

            for link in self.graph.get_neighbors(current_name):
                if link.max_link_capacity <= 0:
                    continue
                neighbor_name = link.get_destination(current_name)
                neighbor_hub = self.graph.nodes_dict.get(neighbor_name)
                if neighbor_hub is None or neighbor_hub.zone == HubType.BLOCKED:
                    continue
                if neighbor_name != start_hub.name and neighbor_hub.max_drones <= 0:
                    continue

                new_cost = current_cost + step_cost
                if new_cost < dist.get(neighbor_name, float("inf")):
                    dist[neighbor_name] = new_cost
                    heapq.heappush(pq, (new_cost, neighbor_name))

        self.heuristics = dist
        if start_hub.name not in self.heuristics:
            raise ValueError(
                f"No valid route exists between '{start_hub.name}' "
                f"and '{end_hub.name}'."
            )

    def _can_occupy_hub(self, hub_name: str, turn: int) -> bool:
        """Check whether a hub has capacity available on the given turn."""
        start_hub, end_hub = self._validate_graph()
        if hub_name in (start_hub.name, end_hub.name):
            return True
        hub = self.graph.nodes_dict[hub_name]
        return self.hub_reservations[hub_name][turn] < hub.max_drones

    def _can_use_link(self, link: Connection, turn: int) -> bool:
        """Check whether a connection has capacity available on the given turn."""
        canon_id = self._canonical_link_id(link)
        return self.link_reservations[canon_id][turn] < link.max_link_capacity

    def _reserve_trajectory(self, steps: tuple[ScheduledStep, ...]) -> None:
        """Record the hubs and connections used by a drone."""
        start_hub, end_hub = self._validate_graph()
        for step in steps:
            if not step.is_wait:
                canon_id = self._canonical_link_id(step.connection)
                self.link_reservations[canon_id][step.turn] += 1
            if not step.is_restricted_transit:
                if step.to_zone not in (start_hub.name, end_hub.name):
                    self.hub_reservations[step.to_zone][step.turn] += 1

    def _find_optimal_trajectory_for_drone(self) -> tuple[ScheduledStep, ...]:
        """Find a collision-free route while respecting earlier reservations."""
        start_hub, end_hub = self._validate_graph()
        start_h = self.heuristics[start_hub.name]

        initial_node = _SearchNode(
            f_score=(start_h, 0, 0, 0, start_hub.name),
            g_turn=0,
            current_zone=start_hub.name,
            departure_turn=None,
            priority_count=0,
            hops=0,
            steps=(),
        )

        open_heap: list[_SearchNode] = [initial_node]
        # Keep the best (-priority_count, active_time, hops) for each zone and turn.
        best_seen: dict[tuple[str, int], tuple[int, int, int]] = {}

        while open_heap:
            node = heapq.heappop(open_heap)
            u = node.current_zone
            t = node.g_turn

            if u == end_hub.name:
                return node.steps

            if t >= self.max_turns_limit:
                continue

            active_time = (
                0 if node.departure_turn is None else t - node.departure_turn
            )
            state_key = (u, t)
            state_quality = (-node.priority_count, active_time, node.hops)
            if state_key in best_seen and best_seen[state_key] <= state_quality:
                continue
            best_seen[state_key] = state_quality

            # 1. Explore moves along neighboring connections.
            for link in self.graph.get_neighbors(u):
                if link.max_link_capacity <= 0:
                    continue
                v = link.get_destination(u)
                if v not in self.heuristics:
                    continue

                v_hub = self.graph.nodes_dict[v]
                move_cost = 2 if v_hub.zone == HubType.RESTRICTED else 1
                arrival_turn = t + move_cost

                if move_cost == 1:
                    if not self._can_use_link(link, t + 1):
                        continue
                    if not self._can_occupy_hub(v, t + 1):
                        continue
                    oriented = self._orient_connection(link, u, v)
                    new_steps = node.steps + (
                        ScheduledStep(
                            turn=t + 1,
                            from_zone=u,
                            to_zone=v,
                            connection=oriented,
                        ),
                    )
                else:
                    # Restricted transit uses t+1 and reaches the hub on t+2.
                    if not (
                        self._can_use_link(link, t + 1)
                        and self._can_use_link(link, t + 2)
                    ):
                        continue
                    if not self._can_occupy_hub(v, t + 2):
                        continue
                    oriented = self._orient_connection(link, u, v)
                    new_steps = node.steps + (
                        ScheduledStep(
                            turn=t + 1,
                            from_zone=u,
                            to_zone=v,
                            connection=oriented,
                            is_restricted_transit=True,
                        ),
                        ScheduledStep(
                            turn=t + 2,
                            from_zone=u,
                            to_zone=v,
                            connection=oriented,
                        ),
                    )

                new_departure = (
                    t if node.departure_turn is None else node.departure_turn
                )
                new_priority = node.priority_count + (
                    1 if v_hub.zone == HubType.PRIORITY else 0
                )
                new_hops = node.hops + 1
                est_arrival = arrival_turn + self.heuristics[v]
                est_active = est_arrival - new_departure

                next_node = _SearchNode(
                    f_score=(est_arrival, -new_priority, est_active, new_hops, v),
                    g_turn=arrival_turn,
                    current_zone=v,
                    departure_turn=new_departure,
                    priority_count=new_priority,
                    hops=new_hops,
                    steps=new_steps,
                )
                heapq.heappush(open_heap, next_node)

            # 2. Explore waiting for one turn at the current hub.
            if self._can_occupy_hub(u, t + 1):
                wait_conn = Connection(from_zone=u, to_zone=u, max_link_capacity=0)
                new_steps = node.steps + (
                    ScheduledStep(
                        turn=t + 1,
                        from_zone=u,
                        to_zone=u,
                        connection=wait_conn,
                        is_wait=True,
                    ),
                )
                if node.departure_turn is None and u == start_hub.name:
                    dep = None
                    est_active = self.heuristics[u]
                else:
                    dep = 0 if node.departure_turn is None else node.departure_turn
                    est_active = (t + 1) + self.heuristics[u] - dep

                est_arrival = (t + 1) + self.heuristics[u]
                wait_node = _SearchNode(
                    f_score=(
                        est_arrival,
                        -node.priority_count,
                        est_active,
                        node.hops,
                        u,
                    ),
                    g_turn=t + 1,
                    current_zone=u,
                    departure_turn=dep,
                    priority_count=node.priority_count,
                    hops=node.hops,
                    steps=new_steps,
                )
                heapq.heappush(open_heap, wait_node)

        raise RuntimeError(
            "Could not find a collision-free route within the turn limit."
        )

"""Public solver API for planning collision-free drone routes."""
from __future__ import annotations

from collections import defaultdict

from Backend.Drones.drone import Drone
from Backend.Solver.models import ScheduledStep
from Backend.Solver.pathfinding import _Pathfinding
from Backend.Solver.solution_output import _SolutionOutput
from Backend.Zones.conection import Connection
from Backend.Zones.graph import Graph
from Backend.Zones.hub import Hub
from Backend.Zones.hub_type import HubType

__all__ = [
    "Connection",
    "Drone",
    "Graph",
    "Hub",
    "HubType",
    "ScheduledStep",
    "Solver",
]


class Solver(_Pathfinding, _SolutionOutput):
    """
    Plan optimal routes for multiple drones on a Fly-In graph.

    Combines reverse Dijkstra heuristics with space-time A* and capacity
    reservations for hubs and connections.
    """

    def __init__(self, graph: Graph, max_turns_limit: int = 5000) -> None:
        self.graph: Graph = graph
        self.max_turns_limit: int = max_turns_limit
        self.heuristics: dict[str, int] = {}
        self.drones: list[Drone] = []
        self.schedules: dict[str, list[ScheduledStep]] = {}
        self.hub_reservations: dict[str, dict[int, int]] = defaultdict(
            lambda: defaultdict(int)
        )
        self.link_reservations: dict[str, dict[int, int]] = defaultdict(
            lambda: defaultdict(int)
        )
        self.total_turns: int = 0

    def solve(
        self,
        include_waits_in_path: bool = False,
        drone_prefix: str = "D",
    ) -> list[Drone]:
        """
        Find an optimal path for each drone and return the created drones.

        Args:
            include_waits_in_path: Include wait steps and restricted-transit
                steps in each drone's path. When false, paths contain only
                completed connections from start to end.
            drone_prefix: Prefix for drone IDs, such as ``"D"`` for ``"D1"``.
        """
        self._validate_graph()
        self.drones.clear()
        self.schedules.clear()
        self.hub_reservations.clear()
        self.link_reservations.clear()
        self.total_turns = 0

        if self.graph.n_drones == 0:
            return []

        self._compute_backward_heuristics()

        for i in range(1, self.graph.n_drones + 1):
            drone_id = f"{drone_prefix}{i}"
            trajectory = self._find_optimal_trajectory_for_drone()
            self._reserve_trajectory(trajectory)
            self.schedules[drone_id] = list(trajectory)

            if trajectory:
                self.total_turns = max(self.total_turns, trajectory[-1].turn)

            if include_waits_in_path:
                drone_path = [step.connection for step in trajectory]
            else:
                drone_path = [
                    step.connection
                    for step in trajectory
                    if not step.is_wait and not step.is_restricted_transit
                ]

            self.drones.append(Drone(id=drone_id, path=drone_path))

        return self.drones

"""Coordinate map loading, route planning, output, and optional visualization."""
from __future__ import annotations

from pathlib import Path

from Backend.Parser.map_parser import MapParser
from Backend.Drones.drone import Drone
from Backend.Solver.solver import Solver
from Backend.Zones.graph import Graph


class Orchestrator:
    """Run the central Fly-In application workflow."""

    def run(self, map_file: str | Path, use_gui: bool = True) -> None:
        """Load a map, plan drone routes, print results, and optionally show the GUI."""
        content = MapParser(file_name=str(map_file))
        graph = self._build_graph(content)
        solver = Solver(graph)
        drones = solver.solve()
        self._print_results(graph, solver, drones)

        if use_gui:
            from Frontend.renderer import Renderer

            print("\nInitializing visualizer...")
            Renderer(graph=graph, solver=solver).run()

    @staticmethod
    def _build_graph(content: MapParser) -> Graph:
        graph = Graph()
        for hub in content.hubs:
            graph.add_node(hub)
        for connection in content.connections:
            graph.add_link(connection)
        graph.n_drones = content.nb_drones
        return graph

    @staticmethod
    def _print_results(graph: Graph, solver: Solver, drones: list[Drone]) -> None:
        start_name = graph.start.name if graph.start else "start"
        end_name = graph.end.name if graph.end else "goal"

        print(f"=== Optimal routes assigned ({graph.n_drones} drones) ===")
        for drone in drones:
            route_nodes = [start_name] + [
                connection.to_zone for connection in drone.path
            ]
            route = " -> ".join(route_nodes)
            schedule = solver.schedules.get(drone.id, [])
            arrival_turn = schedule[-1].turn if schedule else 0
            print(f"{drone.id}: {route} (arrives on turn {arrival_turn})")

        print("\n=== Turn-by-turn simulation ===")
        simulation_output = solver.format_solution(show_turn_prefix=True)
        print(
            simulation_output
            if simulation_output
            else "No moves are required."
        )

        print("\n=== Summary ===")
        print(
            f"Total turns to reach the goal ('{end_name}'): {solver.total_turns}"
        )

"""Minimalist single-file Pygame renderer for Fly-In."""
from __future__ import annotations

import math
import pygame

from Backend.Solver.models import ScheduledStep
from Backend.Solver.solver import Solver
from Backend.Zones.graph import Graph

COLORS: dict[str, tuple[int, int, int]] = {
    "white": (235, 240, 245),
    "black": (60, 70, 90),
    "red": (235, 85, 95),
    "darkred": (180, 50, 65),
    "crimson": (220, 60, 90),
    "maroon": (150, 55, 85),
    "green": (80, 215, 150),
    "blue": (85, 170, 245),
    "yellow": (245, 215, 95),
    "gold": (240, 190, 70),
    "orange": (245, 145, 80),
    "purple": (175, 115, 240),
    "violet": (205, 135, 245),
    "brown": (185, 130, 95),
    "cyan": (90, 225, 235),
    "gray": (150, 165, 185),
    "rainbow": (255, 120, 220),
}


class Renderer:
    """Minimalist turn-based visualizer that scales dynamically with the screen."""

    def __init__(self, graph: Graph, solver: Solver) -> None:
        self.graph = graph
        self.solver = solver
        self.turn = 0
        self.fullscreen = False
        self.show_names = True
        self.win_size = (1200, 800)

        self.steps_by_drone: dict[str, dict[int, ScheduledStep]] = {}
        self.last_turn: dict[str, int] = {}
        for drone in solver.drones:
            sched = solver.schedules.get(drone.id, [])
            self.steps_by_drone[drone.id] = {s.turn: s for s in sched}
            self.last_turn[drone.id] = sched[-1].turn if sched else 0

    def _to_screen(
        self, x: float, y: float, width: int, height: int
    ) -> tuple[int, int]:
        """Map world coordinates (x, y) to screen pixels with aspect-ratio scaling."""
        if not self.graph.nodes_dict:
            return width // 2, height // 2

        xs = [h.x for h in self.graph.nodes_dict.values()]
        ys = [h.y for h in self.graph.nodes_dict.values()]
        min_x, max_x = float(min(xs)), float(max(xs))
        min_y, max_y = float(min(ys)), float(max(ys))

        span_x = max(1.0, max_x - min_x)
        span_y = max(1.0, max_y - min_y)

        pad_x = max(70.0, width * 0.12)
        pad_y = max(70.0, height * 0.12)
        usable_w = max(100.0, width - 2.0 * pad_x)
        usable_h = max(100.0, height - 2.0 * pad_y)

        scale = min(usable_w / span_x, usable_h / span_y)
        cx, cy = width * 0.5, height * 0.5
        mid_x, mid_y = (min_x + max_x) * 0.5, (min_y + max_y) * 0.5

        sx = int(round(cx + (x - mid_x) * scale))
        sy = int(round(cy - (y - mid_y) * scale))
        return sx, sy

    def _drone_world_pos(self, drone_id: str) -> tuple[float, float]:
        """Return world coordinates (wx, wy) of a drone on the current turn."""
        start = self.graph.start
        end = self.graph.end
        if start is None or end is None:
            return 0.0, 0.0

        if self.turn <= 0:
            return float(start.x), float(start.y)

        end_t = self.last_turn.get(drone_id, 0)
        if end_t == 0 or self.turn >= end_t:
            return float(end.x), float(end.y)

        step = self.steps_by_drone.get(drone_id, {}).get(self.turn)
        if step is None:
            return float(start.x), float(start.y)

        if step.is_restricted_transit:
            u = self.graph.nodes_dict[step.from_zone]
            v = self.graph.nodes_dict[step.to_zone]
            return (u.x + v.x) * 0.5, (u.y + v.y) * 0.5

        hub = self.graph.nodes_dict[step.to_zone]
        return float(hub.x), float(hub.y)

    @staticmethod
    def _place_label(
        pos: tuple[int, int],
        size: tuple[int, int],
        radius: int,
        occupied: list[pygame.Rect],
    ) -> pygame.Rect:
        """Find a non-overlapping bounding box for a node label."""
        px, py = pos
        tw, th = size
        bw, bh = tw + 8, th + 4

        dirs = [
            (0.0, 1.0),
            (0.0, -1.0),
            (1.0, 0.0),
            (-1.0, 0.0),
            (0.85, 0.85),
            (-0.85, 0.85),
            (0.85, -0.85),
            (-0.85, -0.85),
        ]

        best_rect = pygame.Rect(0, 0, bw, bh)
        best_cost = float("inf")

        for ring in (1.0, 1.85, 2.7):
            dist_x = (radius + bw * 0.5 + 5.0) * ring
            dist_y = (radius + bh * 0.5 + 5.0) * ring
            for dx, dy in dirs:
                cx = int(round(px + dx * dist_x))
                cy = int(round(py + dy * dist_y))
                cand = pygame.Rect(cx - bw // 2, cy - bh // 2, bw, bh)

                overlap = 0
                for occ in occupied:
                    if cand.colliderect(occ):
                        ix = min(cand.right, occ.right) - max(cand.left, occ.left)
                        iy = min(cand.bottom, occ.bottom) - max(cand.top, occ.top)
                        overlap += ix * iy

                if overlap == 0:
                    return cand

                cost = overlap + ring * 25.0
                if cost < best_cost:
                    best_cost = cost
                    best_rect = cand

        return best_rect

    def draw_edges(self, screen: pygame.Surface) -> None:
        """Draw graph connections."""
        w, h = screen.get_size()
        for link in self.graph.link_dict.values():
            u = self.graph.nodes_dict[link.from_zone]
            v = self.graph.nodes_dict[link.to_zone]
            p1 = self._to_screen(u.x, u.y, w, h)
            p2 = self._to_screen(v.x, v.y, w, h)
            pygame.draw.line(screen, (85, 95, 120), p1, p2, 2)

    def draw_nodes(
        self, screen: pygame.Surface, font: pygame.font.Font
    ) -> None:
        """Draw hubs and optionally their non-overlapping labels."""
        w, h = screen.get_size()
        radius = max(8, int(min(w, h) * 0.018))

        occupied: list[pygame.Rect] = []
        node_data: list[tuple[tuple[int, int], tuple[int, int, int], str]] = []

        for hub in self.graph.nodes_dict.values():
            pos = self._to_screen(hub.x, hub.y, w, h)
            color = COLORS.get(hub.color.strip().lower(), COLORS["white"])
            pygame.draw.circle(screen, color, pos, radius)
            pygame.draw.circle(screen, (20, 24, 34), pos, radius, 2)

            if self.show_names:
                occupied.append(
                    pygame.Rect(
                        pos[0] - radius - 2,
                        pos[1] - radius - 2,
                        (radius + 2) * 2,
                        (radius + 2) * 2,
                    )
                )
                node_data.append((pos, color, hub.name))

        if not self.show_names:
            return

        for pos, color, name in node_data:
            label = font.render(name, True, (235, 240, 250))
            box = self._place_label(pos, label.get_size(), radius, occupied)
            occupied.append(box)

            if math.hypot(box.centerx - pos[0], box.centery - pos[1]) > radius * 2.4:
                pygame.draw.line(screen, (95, 105, 130), pos, box.center, 1)

            pygame.draw.rect(screen, (18, 22, 32), box, border_radius=4)
            pygame.draw.rect(screen, color, box, width=1, border_radius=4)
            screen.blit(label, label.get_rect(center=box.center))

    def draw_drones(
        self, screen: pygame.Surface, font: pygame.font.Font
    ) -> None:
        """Draw drones with lowercase labels (d1, d2, d3...)."""
        w, h = screen.get_size()
        radius = max(5, int(min(w, h) * 0.010))

        groups: dict[tuple[int, int], list[str]] = {}
        for drone in self.solver.drones:
            wx, wy = self._drone_world_pos(drone.id)
            spos = self._to_screen(wx, wy, w, h)
            groups.setdefault(spos, []).append(drone.id)

        for (bx, by), drone_ids in groups.items():
            count = len(drone_ids)
            spread = max(12, int(radius * 2.2)) if count > 1 else 0
            for idx, did in enumerate(drone_ids):
                if count > 1:
                    angle = (2.0 * math.pi * idx) / count
                    dx = int(round(bx + math.cos(angle) * spread))
                    dy = int(round(by + math.sin(angle) * spread))
                else:
                    dx, dy = bx, by

                pygame.draw.circle(screen, (0, 230, 255), (dx, dy), radius)
                pygame.draw.circle(screen, (15, 20, 30), (dx, dy), radius, 1)

                label = font.render(did.lower(), True, (255, 240, 150))
                rect = label.get_rect(center=(dx, dy - radius - 9))
                screen.blit(label, rect)

    def draw_turn(
        self, screen: pygame.Surface, font: pygame.font.Font
    ) -> None:
        """Render the current turn counter on screen."""
        text = f"Turn: {self.turn} / {self.solver.total_turns}"
        surf = font.render(text, True, (240, 245, 255))
        screen.blit(surf, (20, 20))

    def run(self) -> None:
        """Run the main event and rendering loop."""
        pygame.init()
        pygame.display.set_caption("Fly-In")
        screen = pygame.display.set_mode(self.win_size, pygame.RESIZABLE)
        clock = pygame.time.Clock()

        running = True
        while running:
            clock.tick(60)
            for event in pygame.event.get():
                if event.type == pygame.QUIT:
                    running = False
                elif event.type == pygame.VIDEORESIZE and not self.fullscreen:
                    self.win_size = (event.w, event.h)
                    screen = pygame.display.set_mode(
                        self.win_size, pygame.RESIZABLE
                    )
                elif event.type == pygame.KEYDOWN:
                    if event.key == pygame.K_ESCAPE:
                        running = False
                    elif event.key == pygame.K_RIGHT:
                        self.turn = min(self.solver.total_turns, self.turn + 1)
                    elif event.key == pygame.K_LEFT:
                        self.turn = max(0, self.turn - 1)
                    elif event.key == pygame.K_n:
                        self.show_names = not self.show_names
                    elif event.key == pygame.K_f:
                        self.fullscreen = not self.fullscreen
                        if self.fullscreen:
                            screen = pygame.display.set_mode(
                                (0, 0), pygame.FULLSCREEN
                            )
                        else:
                            screen = pygame.display.set_mode(
                                self.win_size, pygame.RESIZABLE
                            )

            w, h = screen.get_size()
            font_size = max(11, int(min(w, h) * 0.016))
            font = pygame.font.SysFont("monospace", font_size, bold=True)

            screen.fill((18, 22, 32))
            self.draw_edges(screen)
            self.draw_nodes(screen, font)
            self.draw_drones(screen, font)
            self.draw_turn(screen, font)
            pygame.display.flip()

        pygame.quit()

#!/usr/bin/env python3
"""Command-line entry point for the Fly-In application."""
from __future__ import annotations

import argparse
import sys
from collections.abc import Sequence


def main(argv: Sequence[str] | None = None) -> int:
    parser = argparse.ArgumentParser(
        description="Plan routes for multiple drones and optionally visualize them."
    )
    parser.add_argument("map_file", help="Path to the map file.")
    parser.add_argument(
        "--no-gui",
        action="store_true",
        help="Run route planning without opening the Pygame visualizer.",
    )
    args = parser.parse_args(argv)
    try:
        from Backend.orchestrator import Orchestrator

        Orchestrator().run(args.map_file, use_gui=not args.no_gui)
    except (ValueError, FileNotFoundError, PermissionError, RuntimeError, IsADirectoryError) as error:
        print(error, file=sys.stderr)
        return 1
    return 0


if __name__ == "__main__":
    try:
        sys.exit(main())
    except KeyboardInterrupt:
        print("\nThe program has been interrupted.", file=sys.stderr)
        sys.exit(130)

        print(e)
