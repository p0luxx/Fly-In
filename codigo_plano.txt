#!/bin/usr/python3
"""Command-line entry point for the Fly-In application."""
from __future__ import annotations

import argparse
import sys
from collections.abc import Sequence


def main(argv: Sequence[str] | None = None) -> int:
    parser = argparse.ArgumentParser(
        description="Plan routes for multiple drones and optionally visualize them."
    )
    parser.add_argument("map_file", help="Path to the map file.")
    parser.add_argument(
        "--no-gui",
        action="store_true",
        help="Run route planning without opening the Pygame visualizer.",
    )
    args = parser.parse_args(argv)

    try:
        from Backend.orchestrator import Orchestrator

        Orchestrator().run(args.map_file, use_gui=not args.no_gui)
    except (ValueError, FileNotFoundError, PermissionError, RuntimeError) as error:
        print(error, file=sys.stderr)
        return 1
    return 0


if __name__ == "__main__":
    try:
        sys.exit(main())
    except KeyboardInterrupt:
        print("\nThe program has been interrupted.", file=sys.stderr)
        sys.exit(130)

class Drone:
    def __init__(self, id: str, path: list[Connection]) -> None:
        self.id = id
        self.path = path
        self.steps: Generator[Connection, None, None] = self._next_step()
        self.c_pos: Connection | None = None

    def _next_step(self) -> Generator[Connection, None, None]:
        for step in self.path:
            self.c_pos = step
            yield step

"""Load and validate map files for the Fly-In application."""
from __future__ import annotations

from typing import Any

from pydantic import BaseModel, Field, model_validator

from Backend.Zones.conection import Connection
from Backend.Zones.hub import Hub
from Backend.Zones.hub_type import HubType


class MapParser(BaseModel):
    file_name: str
    content: list[str] = Field(default_factory=list, exclude=True)
    nb_drones: int = 0
    hubs: list[Hub] = Field(default_factory=list)
    connections: list[Connection] = Field(default_factory=list)

    @model_validator(mode="after")
    def process_file_automatically(self) -> MapParser:
        self._read_file()
        self._parse_and_route()
        start_count = sum(hub.directive == "start_hub" for hub in self.hubs)
        end_count = sum(hub.directive == "end_hub" for hub in self.hubs)
        if start_count != 1 or end_count != 1:
            raise ValueError(
                "The map must contain exactly one start_hub and one end_hub."
            )
        return self

    def _read_file(self) -> None:
        try:
            with open(self.file_name, "r", encoding="utf-8") as map_file:
                raw_lines = map_file.readlines()
        except PermissionError as error:
            raise PermissionError(
                f"Permission denied while reading map file: {self.file_name}"
            ) from error
        except FileNotFoundError as error:
            raise FileNotFoundError(
                f"Map file not found: {self.file_name}"
            ) from error

        self.content = [
            line.strip()
            for line in raw_lines
            if line.strip() and not line.strip().startswith("#")
        ]

    def _parse_and_route(self) -> None:
        try:
            for line in self.content:
                if ":" not in line:
                    raise ValueError("Invalid map syntax: missing ':' separator.")
                key, value = (part.strip() for part in line.split(":", 1))
                match key:
                    case "nb_drones":
                        self._drone_processing(value)
                    case "start_hub" | "hub" | "end_hub":
                        self._hub_processor(key, value)
                    case "connection":
                        self._connection_processor(value)
                    case _:
                        raise ValueError(f"Unknown map directive: {key}")
        except ValueError as error:
            raise ValueError(f"Invalid map: {error}") from error

    def _drone_processing(self, value: str) -> None:
        self.nb_drones = int(value)

    def _hub_processor(self, directive: str, value: str) -> None:
        parts = value.split("[", 1)
        coordinates = parts[0].split()
        if len(coordinates) != 3:
            raise ValueError("A hub must define a name and x/y coordinates.")
        name, x_str, y_str = coordinates

        extras: dict[str, Any] = {}
        if len(parts) > 1:
            extra_raw = parts[1].strip()
            if not extra_raw.endswith("]"):
                raise ValueError("Invalid hub syntax: missing closing ']'.")
            for item in extra_raw[:-1].split():
                key, value_part = item.split("=", 1)
                try:
                    if value_part.lstrip("-").isnumeric():
                        extras[key] = int(value_part)
                    elif key == "zone":
                        extras[key] = HubType[value_part.upper()]
                    else:
                        extras[key] = value_part
                except KeyError as error:
                    raise ValueError(
                        "Invalid hub zone type; use a supported HubType."
                    ) from error

        self.hubs.append(
            Hub(
                directive=directive,
                name=name,
                x=int(x_str),
                y=int(y_str),
                **extras,
            )
        )

    def _connection_processor(self, value: str) -> None:
        parts = value.strip().split("[", 1)
        route_str = parts[0].strip()
        if "-" not in route_str:
            raise ValueError(
                f"Invalid connection syntax (missing '-'): '{route_str}'"
            )

        from_zone, to_zone = (part.strip() for part in route_str.split("-", 1))
        if not from_zone or not to_zone:
            raise ValueError(f"Connection endpoints cannot be empty: '{route_str}'")

        capacity = 1
        if len(parts) > 1:
            extra_raw = parts[1].strip()
            if not extra_raw.endswith("]"):
                raise ValueError("Invalid connection syntax: missing closing ']'.")
            for item in extra_raw[:-1].split():
                if "=" not in item:
                    continue
                key, raw_value = item.split("=", 1)
                if key == "max_link_capacity":
                    if not raw_value.isdigit() or int(raw_value) <= 0:
                        raise ValueError(
                            "max_link_capacity must be a positive integer."
                        )
                    capacity = int(raw_value)

        self.connections.append(
            Connection(
                from_zone=from_zone,
                to_zone=to_zone,
                max_link_capacity=capacity,
            )
        )

"""Graph validation, capacity reservations, and space-time route search."""
from __future__ import annotations

import heapq

from Backend.Solver.models import ScheduledStep, _SearchNode
from Backend.Zones.conection import Connection
from Backend.Zones.hub import Hub
from Backend.Zones.hub_type import HubType
from Backend.Zones.graph import Graph

__all__ = ["_Pathfinding"]


class _Pathfinding:
    """Route-search behavior mixed into the public solver."""

    graph: Graph
    heuristics: dict[str, int]
    hub_reservations: dict[str, dict[int, int]]
    link_reservations: dict[str, dict[int, int]]
    max_turns_limit: int

    def _validate_graph(self) -> tuple[Hub, Hub]:
        if self.graph.start is None or self.graph.end is None:
            raise ValueError(
                "The graph must define both a 'start_hub' and an 'end_hub'."
            )
        if self.graph.n_drones < 0:
            raise ValueError("The number of drones cannot be negative.")
        if (
            self.graph.start.zone == HubType.BLOCKED
            or self.graph.end.zone == HubType.BLOCKED
        ):
            raise ValueError("The start or end hub is blocked (BLOCKED).")
        return self.graph.start, self.graph.end

    @staticmethod
    def _canonical_link_id(link: Connection) -> str:
        """Return a direction-independent identifier for a connection."""
        a, b = sorted((link.from_zone, link.to_zone))
        return f"{a}<->{b}"

    @staticmethod
    def _orient_connection(
        link: Connection, from_zone: str, to_zone: str
    ) -> Connection:
        """Orient a connection in the direction of travel."""
        if link.from_zone == from_zone and link.to_zone == to_zone:
            return link
        return Connection(
            from_zone=from_zone,
            to_zone=to_zone,
            max_link_capacity=link.max_link_capacity,
        )

    def _compute_backward_heuristics(self) -> None:
        """Compute the admissible minimum-turn distance to the end hub."""
        start_hub, end_hub = self._validate_graph()
        dist: dict[str, int] = {end_hub.name: 0}
        pq: list[tuple[int, str]] = [(0, end_hub.name)]

        while pq:
            current_cost, current_name = heapq.heappop(pq)
            if current_cost > dist.get(current_name, float("inf")):
                continue

            current_hub = self.graph.nodes_dict[current_name]
            step_cost = 2 if current_hub.zone == HubType.RESTRICTED else 1

            for link in self.graph.get_neighbors(current_name):
                if link.max_link_capacity <= 0:
                    continue
                neighbor_name = link.get_destination(current_name)
                neighbor_hub = self.graph.nodes_dict.get(neighbor_name)
                if neighbor_hub is None or neighbor_hub.zone == HubType.BLOCKED:
                    continue
                if neighbor_name != start_hub.name and neighbor_hub.max_drones <= 0:
                    continue

                new_cost = current_cost + step_cost
                if new_cost < dist.get(neighbor_name, float("inf")):
                    dist[neighbor_name] = new_cost
                    heapq.heappush(pq, (new_cost, neighbor_name))

        self.heuristics = dist
        if start_hub.name not in self.heuristics:
            raise ValueError(
                f"No valid route exists between '{start_hub.name}' "
                f"and '{end_hub.name}'."
            )

    def _can_occupy_hub(self, hub_name: str, turn: int) -> bool:
        """Check whether a hub has capacity available on the given turn."""
        start_hub, end_hub = self._validate_graph()
        if hub_name in (start_hub.name, end_hub.name):
            return True
        hub = self.graph.nodes_dict[hub_name]
        return self.hub_reservations[hub_name][turn] < hub.max_drones

    def _can_use_link(self, link: Connection, turn: int) -> bool:
        """Check whether a connection has capacity available on the given turn."""
        canon_id = self._canonical_link_id(link)
        return self.link_reservations[canon_id][turn] < link.max_link_capacity

    def _reserve_trajectory(self, steps: tuple[ScheduledStep, ...]) -> None:
        """Record the hubs and connections used by a drone."""
        start_hub, end_hub = self._validate_graph()
        for step in steps:
            if not step.is_wait:
                canon_id = self._canonical_link_id(step.connection)
                self.link_reservations[canon_id][step.turn] += 1
            if not step.is_restricted_transit:
                if step.to_zone not in (start_hub.name, end_hub.name):
                    self.hub_reservations[step.to_zone][step.turn] += 1

    def _find_optimal_trajectory_for_drone(self) -> tuple[ScheduledStep, ...]:
        """Find a collision-free route while respecting earlier reservations."""
        start_hub, end_hub = self._validate_graph()
        start_h = self.heuristics[start_hub.name]

        initial_node = _SearchNode(
            f_score=(start_h, 0, 0, 0, start_hub.name),
            g_turn=0,
            current_zone=start_hub.name,
            departure_turn=None,
            priority_count=0,
            hops=0,
            steps=(),
        )

        open_heap: list[_SearchNode] = [initial_node]
        # Keep the best (-priority_count, active_time, hops) for each zone and turn.
        best_seen: dict[tuple[str, int], tuple[int, int, int]] = {}

        while open_heap:
            node = heapq.heappop(open_heap)
            u = node.current_zone
            t = node.g_turn

            if u == end_hub.name:
                return node.steps

            if t >= self.max_turns_limit:
                continue

            active_time = (
                0 if node.departure_turn is None else t - node.departure_turn
            )
            state_key = (u, t)
            state_quality = (-node.priority_count, active_time, node.hops)
            if state_key in best_seen and best_seen[state_key] <= state_quality:
                continue
            best_seen[state_key] = state_quality

            # 1. Explore moves along neighboring connections.
            for link in self.graph.get_neighbors(u):
                if link.max_link_capacity <= 0:
                    continue
                v = link.get_destination(u)
                if v not in self.heuristics:
                    continue

                v_hub = self.graph.nodes_dict[v]
                move_cost = 2 if v_hub.zone == HubType.RESTRICTED else 1
                arrival_turn = t + move_cost

                if move_cost == 1:
                    if not self._can_use_link(link, t + 1):
                        continue
                    if not self._can_occupy_hub(v, t + 1):
                        continue
                    oriented = self._orient_connection(link, u, v)
                    new_steps = node.steps + (
                        ScheduledStep(
                            turn=t + 1,
                            from_zone=u,
                            to_zone=v,
                            connection=oriented,
                        ),
                    )
                else:
                    # Restricted transit uses t+1 and reaches the hub on t+2.
                    if not (
                        self._can_use_link(link, t + 1)
                        and self._can_use_link(link, t + 2)
                    ):
                        continue
                    if not self._can_occupy_hub(v, t + 2):
                        continue
                    oriented = self._orient_connection(link, u, v)
                    new_steps = node.steps + (
                        ScheduledStep(
                            turn=t + 1,
                            from_zone=u,
                            to_zone=v,
                            connection=oriented,
                            is_restricted_transit=True,
                        ),
                        ScheduledStep(
                            turn=t + 2,
                            from_zone=u,
                            to_zone=v,
                            connection=oriented,
                        ),
                    )

                new_departure = (
                    t if node.departure_turn is None else node.departure_turn
                )
                new_priority = node.priority_count + (
                    1 if v_hub.zone == HubType.PRIORITY else 0
                )
                new_hops = node.hops + 1
                est_arrival = arrival_turn + self.heuristics[v]
                est_active = est_arrival - new_departure

                next_node = _SearchNode(
                    f_score=(est_arrival, -new_priority, est_active, new_hops, v),
                    g_turn=arrival_turn,
                    current_zone=v,
                    departure_turn=new_departure,
                    priority_count=new_priority,
                    hops=new_hops,
                    steps=new_steps,
                )
                heapq.heappush(open_heap, next_node)

            # 2. Explore waiting for one turn at the current hub.
            if self._can_occupy_hub(u, t + 1):
                wait_conn = Connection(from_zone=u, to_zone=u, max_link_capacity=0)
                new_steps = node.steps + (
                    ScheduledStep(
                        turn=t + 1,
                        from_zone=u,
                        to_zone=u,
                        connection=wait_conn,
                        is_wait=True,
                    ),
                )
                if node.departure_turn is None and u == start_hub.name:
                    dep = None
                    est_active = self.heuristics[u]
                else:
                    dep = 0 if node.departure_turn is None else node.departure_turn
                    est_active = (t + 1) + self.heuristics[u] - dep

                est_arrival = (t + 1) + self.heuristics[u]
                wait_node = _SearchNode(
                    f_score=(
                        est_arrival,
                        -node.priority_count,
                        est_active,
                        node.hops,
                        u,
                    ),
                    g_turn=t + 1,
                    current_zone=u,
                    departure_turn=dep,
                    priority_count=node.priority_count,
                    hops=node.hops,
                    steps=new_steps,
                )
                heapq.heappush(open_heap, wait_node)

        raise RuntimeError(
            "Could not find a collision-free route within the turn limit."
        )

"""Formatting of solver schedules for command-line and textual output."""
from __future__ import annotations

from collections import defaultdict

from Backend.Drones.drone import Drone
from Backend.Zones.graph import Graph

__all__ = ["_SolutionOutput"]


class _SolutionOutput:
    """Schedule formatting behavior mixed into the public solver."""

    schedules: dict
    graph: Graph
    total_turns: int

    def solve(
        self,
        include_waits_in_path: bool = False,
        drone_prefix: str = "D",
    ) -> list[Drone]:
        raise NotImplementedError

    def get_turns_output(self) -> list[str]:
        """
        Generate the standard turn-by-turn Fly-In output:
        - Normal or priority move: `D1-destination`
        - Restricted transit (first turn): `D1-origin-destination`
        - Restricted arrival (second turn): `D1-destination`
        """
        if not self.schedules and self.graph.n_drones > 0:
            self.solve()

        turns_dict: dict[int, list[str]] = defaultdict(list)
        for drone_id, steps in self.schedules.items():
            for step in steps:
                if step.is_wait:
                    continue
                if step.is_restricted_transit:
                    turns_dict[step.turn].append(
                        f"{drone_id}-{step.from_zone}-{step.to_zone}"
                    )
                else:
                    turns_dict[step.turn].append(f"{drone_id}-{step.to_zone}")

        return [
            " ".join(turns_dict[t])
            for t in range(1, self.total_turns + 1)
            if turns_dict[t]
        ]

    def format_solution(self, show_turn_prefix: bool = False) -> str:
        """Return the complete turn-by-turn simulation as a string."""
        lines = self.get_turns_output()
        if show_turn_prefix:
            return "\n".join(
                f"Turn {idx}: {line}" for idx, line in enumerate(lines, start=1)
            )
        return "\n".join(lines)

"""Public solver API for planning collision-free drone routes."""
from __future__ import annotations

from collections import defaultdict

from Backend.Drones.drone import Drone
from Backend.Solver.models import ScheduledStep
from Backend.Solver.pathfinding import _Pathfinding
from Backend.Solver.solution_output import _SolutionOutput
from Backend.Zones.conection import Connection
from Backend.Zones.graph import Graph
from Backend.Zones.hub import Hub
from Backend.Zones.hub_type import HubType

__all__ = [
    "Connection",
    "Drone",
    "Graph",
    "Hub",
    "HubType",
    "ScheduledStep",
    "Solver",
]


class Solver(_Pathfinding, _SolutionOutput):
    """
    Plan optimal routes for multiple drones on a Fly-In graph.

    Combines reverse Dijkstra heuristics with space-time A* and capacity
    reservations for hubs and connections.
    """

    def __init__(self, graph: Graph, max_turns_limit: int = 5000) -> None:
        self.graph: Graph = graph
        self.max_turns_limit: int = max_turns_limit
        self.heuristics: dict[str, int] = {}
        self.drones: list[Drone] = []
        self.schedules: dict[str, list[ScheduledStep]] = {}
        self.hub_reservations: dict[str, dict[int, int]] = defaultdict(
            lambda: defaultdict(int)
        )
        self.link_reservations: dict[str, dict[int, int]] = defaultdict(
            lambda: defaultdict(int)
        )
        self.total_turns: int = 0

    def solve(
        self,
        include_waits_in_path: bool = False,
        drone_prefix: str = "D",
    ) -> list[Drone]:
        """
        Find an optimal path for each drone and return the created drones.

        Args:
            include_waits_in_path: Include wait steps and restricted-transit
                steps in each drone's path. When false, paths contain only
                completed connections from start to end.
            drone_prefix: Prefix for drone IDs, such as ``"D"`` for ``"D1"``.
        """
        self._validate_graph()
        self.drones.clear()
        self.schedules.clear()
        self.hub_reservations.clear()
        self.link_reservations.clear()
        self.total_turns = 0

        if self.graph.n_drones == 0:
            return []

        self._compute_backward_heuristics()

        for i in range(1, self.graph.n_drones + 1):
            drone_id = f"{drone_prefix}{i}"
            trajectory = self._find_optimal_trajectory_for_drone()
            self._reserve_trajectory(trajectory)
            self.schedules[drone_id] = list(trajectory)

            if trajectory:
                self.total_turns = max(self.total_turns, trajectory[-1].turn)

            if include_waits_in_path:
                drone_path = [step.connection for step in trajectory]
            else:
                drone_path = [
                    step.connection
                    for step in trajectory
                    if not step.is_wait and not step.is_restricted_transit
                ]

            self.drones.append(Drone(id=drone_id, path=drone_path))

        return self.drones

from dataclasses import dataclass


@dataclass
class Connection:
    from_zone: str = ""
    to_zone: str = ""
    max_link_capacity: int = 1

    @property
    def id(self) -> str:
        return f"{self.from_zone}-{self.to_zone}"

    def get_destination(self, current_zone: str) -> str:
        if current_zone == self.from_zone:
            return self.to_zone
        if current_zone == self.to_zone:
            return self.from_zone
        raise ValueError(
            f"Zone: {current_zone} is not part of connection {self.id}"
        )

from Backend.Zones.conection import Connection
from Backend.Zones.hub import Hub


class Graph:
    def __init__(self) -> None:
        self.start: Hub | None = None
        self.end: Hub | None = None
        self.n_drones: int = 0
        self.link_dict: dict[str, Connection] = {}
        self.nodes_dict: dict[str, Hub] = {}
        self.adj: dict[str, list[Connection]] = {}

    def test(self):
        print("Link dict:")
        for item in self.link_dict.items():
            print(item)
        print("Nodes dict")
        for item in self.nodes_dict.items():
            print(item)

    def add_node(self, node: Hub) -> None:
        if node.directive == "start_hub":
            self.start = node
        elif node.directive == "end_hub":
            self.end = node
        self.nodes_dict[node.name] = node
        self.adj[node.name] = []

    def add_link(self, link: Connection) -> None:
        if link.from_zone in self.nodes_dict and link.to_zone in self.nodes_dict:
            if link.from_zone != link.to_zone:
                self.link_dict[link.id] = link
                self.adj[link.from_zone].append(link)
                self.adj[link.to_zone].append(link)
            else:
                raise ValueError(
                    "A node connected to itself in a bidirectional manner "
                    "has been detected.\nClosing Fly-In..."
                )
        else:
            raise ValueError(
                "An orphaned node has been detected.\nClosing Fly-In..."
            )

    def get_neighbors(self, zone_name: str) -> list[Connection]:
        return self.adj.get(zone_name, [])

    def _init_simulation_structures(self) -> None:

    def _drone_zone_at_turn(self, drone_id: str, turn_idx: int) -> str:
        start_name = self.graph.start.name if self.graph.start else ""
        end_name = self.graph.end.name if self.graph.end else ""
        if turn_idx <= 0:
            return start_name
        arrival = self.arrival_turns.get(drone_id, 0)
        if arrival > 0 and turn_idx >= arrival:
            return end_name
        step = self.step_map.get(drone_id, {}).get(turn_idx)
        return step.to_zone if step is not None else start_name

    def _compute_drone_kinematics(
        self, drone_id: str, sim_t: float
    ) -> tuple[float, float, float, str, str, bool, bool]:
        """
        Calculate the continuous position (wx, wy), heading angle, group key,
        readable status, and activity flags.
        """
        start_hub = self.graph.start
        end_hub = self.graph.end
        assert start_hub is not None and end_hub is not None

        if sim_t <= 0.0:
            return (
                float(start_hub.x),
                float(start_hub.y),
                -math.pi * 0.5,
                f"hub:{start_hub.name}",
                f"At start ({start_hub.name})",
                False,
                False,
            )

        arrival_t = self.arrival_turns.get(drone_id, 0)
        if arrival_t == 0 or sim_t >= float(arrival_t):
            return (
                float(end_hub.x),
                float(end_hub.y),
                -math.pi * 0.5,
                f"hub:{end_hub.name}",
                f"At goal ({end_hub.name})",
                False,
                False,
            )

        current_turn = int(math.floor(sim_t)) + 1
        raw_alpha = sim_t - math.floor(sim_t)

        d_steps = self.step_map.get(drone_id, {})
        step = d_steps.get(current_turn)

        if step is None or step.is_wait:
            zone_name = (
                step.from_zone
                if step is not None
                else self._drone_zone_at_turn(drone_id, current_turn - 1)
            )
            hub = self.graph.nodes_dict[zone_name]
            is_active = zone_name not in (start_hub.name, end_hub.name)
            status = (
                f"Waiting at {zone_name}"
                if is_active
                else f"Ready at {zone_name}"
            )
            return (
                float(hub.x),
                float(hub.y),
                -math.pi * 0.5,
                f"hub:{zone_name}",
                status,
                is_active,
                False,
            )

        u = self.graph.nodes_dict[step.from_zone]
        v = self.graph.nodes_dict[step.to_zone]
        heading = math.atan2(-(v.y - u.y), float(v.x - u.x))

        # Check whether the drone was moving on the previous turn or will continue
        # on the next one, so it can maintain speed through intermediate hubs.
        prev_step = d_steps.get(current_turn - 1)
        next_step = d_steps.get(current_turn + 1)
        continuous_entry = (
            prev_step is not None
            and not prev_step.is_wait
            and prev_step.to_zone == step.from_zone
        )
        continuous_exit = (
            next_step is not None
            and not next_step.is_wait
            and next_step.from_zone == step.to_zone
        )

        if continuous_entry and continuous_exit:
            # Blend linear and sine easing to preserve momentum through a hub.
            alpha = 0.45 * raw_alpha + 0.55 * ease_in_out_sine(raw_alpha)
        else:
            alpha = ease_in_out_sine(raw_alpha)

        # Case 1: First turn entering a RESTRICTED hub (first half of the edge).
        if step.is_restricted_transit:
            frac = 0.5 * ease_in_out_sine(raw_alpha)
            return (
                u.x + (v.x - u.x) * frac,
                u.y + (v.y - u.y) * frac,
                heading,
                f"edge:{u.name}->{v.name}:1",
                f"Restricted transit {u.name} → {v.name} (1/2)",
                True,
                True,
            )

        # Case 2: Second turn entering a RESTRICTED hub (second half of the edge).
        if prev_step is not None and prev_step.is_restricted_transit:
            frac = 0.5 + 0.5 * ease_in_out_sine(raw_alpha)
            return (
                u.x + (v.x - u.x) * frac,
                u.y + (v.y - u.y) * frac,
                heading,
                f"edge:{u.name}->{v.name}:2",
                f"Landing at {v.name} [RESTRICTED] (2/2)",
                True,
                True,
            )

        # Case 3: Standard transit (1 turn).
        return (
            u.x + (v.x - u.x) * alpha,
            u.y + (v.y - u.y) * alpha,
            heading,
            f"edge:{u.name}->{v.name}",
            f"Flying {u.name} → {v.name}",
            True,
            False,
        )

    def _compute_all_drones_render_state(
        self, elapsed: float
    ) -> tuple[list[DroneRenderState], dict[str, list[str]]]:
        """Calculate drones' screen positions and orbital formations."""
        raw: list[tuple[str, float, float, float, str, str, bool, bool]] = []
        groups: dict[str, list[str]] = {}

        for drone in self.solver.drones:
            wx, wy, heading, gkey, status, is_active, in_rest = (
                self._compute_drone_kinematics(drone.id, self.sim_time)
            )
            raw.append((drone.id, wx, wy, heading, gkey, status, is_active, in_rest))
            groups.setdefault(gkey, []).append(drone.id)

            # Detect arrival at the goal and launch a harmonic shockwave.
            arr_t = self.arrival_turns.get(drone.id, 0)
            if arr_t > 0 and self.sim_time >= float(arr_t):
                if drone.id not in self._triggered_arrivals and self.graph.end:
                    self._triggered_arrivals.add(drone.id)
                    self.shockwaves.append(
                        Shockwave(
                            x=float(self.graph.end.x),
                            y=float(self.graph.end.y),
                            color=self.drone_colors[drone.id],
                        )
                    )
            else:
                self._triggered_arrivals.discard(drone.id)

        states: list[DroneRenderState] = []
        for did, wx, wy, heading, gkey, status, is_active, in_rest in raw:
            sx, sy = self.camera.to_screen_float(wx, wy)
            members = groups[gkey]
            if len(members) > 1:
                idx = members.index(did)
                angle = (2.0 * math.pi * idx) / len(members) + elapsed * 0.85
                orbit_px = min(22.0, 6.5 + len(members) * 1.15)
                sx += math.cos(angle) * orbit_px
                sy += math.sin(angle) * orbit_px

            states.append(
                DroneRenderState(
                    drone_id=did,
                    wx=wx,
                    wy=wy,
                    sx=sx,
                    sy=sy,
                    heading_rad=heading,
                    group_key=gkey,
                    status_text=status,
                    is_active=is_active,
                    in_restricted=in_rest,
                )
            )

            if self.playing and is_active:
                self.trails[did].append((wx, wy))

        return states, groups

                    pygame.draw.circle(surface, col, pts[i], r_pt)

            ix, iy = int(round(st.sx)), int(round(st.sy))
            col = self.drone_colors[st.drone_id]

                surface,
                (ix, iy),
                int(drone_radius * (3.0 if is_focus else 2.2)),

            # Show an amber containment halo while crossing a RESTRICTED hub.
            if st.in_restricted:
                pygame.draw.circle(
                    surface, (255, 165, 80), (ix, iy), drone_radius + 4, width=1
                )

            # Draw a soft, oriented droplet/firefly silhouette while in flight.
            if st.is_active:
                nose_x = ix + int(math.cos(st.heading_rad) * (drone_radius + 3))
                nose_y = iy + int(math.sin(st.heading_rad) * (drone_radius + 3))
                pygame.draw.line(
                    surface, (255, 255, 255), (ix, iy), (nose_x, nose_y), 2

            pygame.draw.circle(surface, (12, 16, 30), (ix, iy), drone_radius + 1)
            pygame.draw.circle(surface, col, (ix, iy), drone_radius)
            pygame.draw.circle(
                surface, (255, 255, 255), (ix, iy), max(2, drone_radius // 2)

                pygame.draw.circle(

"""Pygame window lifecycle, input handling, and frame loop."""
from __future__ import annotations

import math

import pygame

from Frontend.visualizer_support import DroneRenderState, clamp
from Frontend.visualizer_rendering import _VisualizerRendering

__all__ = ["_PygameRuntime"]


class _PygameRuntime(_VisualizerRendering):
    """Technical Pygame runtime mixed into :class:`FlyInVisualizer`."""

    def run(self) -> None:
        pygame.init()
        pygame.display.set_caption(
            f"Fly-In Astral Visualizer — {self.graph.n_drones} Drones | "
            f"{self.solver.total_turns} Turns"
        )
        screen = pygame.display.set_mode((self.width, self.height), pygame.RESIZABLE)
        clock = pygame.time.Clock()

        font_title = pygame.font.SysFont("sans", 16, bold=True)
        font_ui = pygame.font.SysFont("sans", 14, bold=True)
        font_node = pygame.font.SysFont("sans", 13, bold=True)
        font_drone = pygame.font.SysFont("sans", 11, bold=True)
        font_small = pygame.font.SysFont("sans", 12)

        dragging_cam = False
        scrubbing_timeline = False
        last_mouse = (0, 0)
        elapsed = 0.0

        node_screen_pos: dict[str, tuple[int, int]] = {}
        drone_states: list[DroneRenderState] = []

        running = True
        while running:
            dt = clamp(clock.tick(60) / 1000.0, 0.001, 0.05)
            elapsed += dt

            for event in pygame.event.get():
                if event.type == pygame.QUIT:
                    running = False

                elif event.type == pygame.VIDEORESIZE:
                    self.width, self.height = event.w, event.h
                    self.camera.width, self.camera.height = event.w, event.h
                    self._bg_surface = None
                    screen = pygame.display.set_mode(
                        (self.width, self.height), pygame.RESIZABLE
                    )

                elif event.type == pygame.KEYDOWN:
                    if event.key == pygame.K_ESCAPE:
                        running = False
                    elif event.key == pygame.K_SPACE:
                        if self.sim_time >= float(self.solver.total_turns):
                            self.sim_time = 0.0
                            self._triggered_arrivals.clear()
                            for tr in self.trails.values():
                                tr.clear()
                        self.playing = not self.playing
                    elif event.key == pygame.K_r:
                        self.sim_time = 0.0
                        self._triggered_arrivals.clear()
                        for tr in self.trails.values():
                            tr.clear()
                        self.playing = True
                    elif event.key == pygame.K_f:
                        self.camera.fit_graph(self.graph, instant=False)
                    elif event.key == pygame.K_n:
                        self.show_node_names = not self.show_node_names
                    elif event.key == pygame.K_d:
                        self.drone_label_mode = (self.drone_label_mode + 1) % 3
                    elif event.key == pygame.K_l:
                        self.show_node_names = not self.show_node_names
                        self.drone_label_mode = 0 if self.show_node_names else 2
                    elif event.key == pygame.K_RIGHT:
                        self.playing = False
                        self.sim_time = min(
                            float(self.solver.total_turns),
                            math.floor(self.sim_time + 1e-5) + 1.0,
                        )
                    elif event.key == pygame.K_LEFT:
                        self.playing = False
                        self.sim_time = max(
                            0.0, math.ceil(self.sim_time - 1e-5) - 1.0
                        )
                        for tr in self.trails.values():
                            tr.clear()
                    elif event.key == pygame.K_UP:
                        self.turns_per_sec = min(8.0, self.turns_per_sec * 1.25)
                    elif event.key == pygame.K_DOWN:
                        self.turns_per_sec = max(0.2, self.turns_per_sec / 1.25)

                elif event.type == pygame.MOUSEBUTTONDOWN:
                    if event.button == 1:
                        if event.pos[1] >= self.height - 72:
                            scrubbing_timeline = True
                            self.playing = False
                            bar_x, bar_w = 42, self.width - 84
                            pct = clamp(
                                (event.pos[0] - bar_x) / max(1, bar_w), 0.0, 1.0
                            )
                            self.sim_time = pct * float(self.solver.total_turns)
                            for tr in self.trails.values():
                                tr.clear()
                        else:
                            # Select a drone or hub for the inspector when clicked.
                            if self.hovered_drone:
                                self.selected_drone = (
                                    None
                                    if self.selected_drone == self.hovered_drone
                                    else self.hovered_drone
                                )
                                self.selected_hub = None
                            elif self.hovered_hub:
                                self.selected_hub = (
                                    None
                                    if self.selected_hub == self.hovered_hub
                                    else self.hovered_hub
                                )
                                self.selected_drone = None
                            else:
                                self.selected_drone = None
                                self.selected_hub = None
                            dragging_cam = True
                            last_mouse = event.pos
                    elif event.button == 4:
                        self.camera.zoom_at(event.pos, 1.14)
                    elif event.button == 5:
                        self.camera.zoom_at(event.pos, 1.0 / 1.14)

                elif event.type == pygame.MOUSEBUTTONUP:
                    if event.button == 1:
                        dragging_cam = False
                        scrubbing_timeline = False

                elif event.type == pygame.MOUSEMOTION:
                    if scrubbing_timeline:
                        bar_x, bar_w = 42, self.width - 84
                        pct = clamp((event.pos[0] - bar_x) / max(1, bar_w), 0.0, 1.0)
                        self.sim_time = pct * float(self.solver.total_turns)
                    elif dragging_cam:
                        dx = event.pos[0] - last_mouse[0]
                        dy = event.pos[1] - last_mouse[1]
                        self.camera.pan(dx, dy)
                        last_mouse = event.pos

            # Update the camera, clock, and expanding shockwaves.
            self.camera.update(dt)

            if self.playing and self.solver.total_turns > 0:
                self.sim_time += dt * self.turns_per_sec
                if self.sim_time >= float(self.solver.total_turns):
                    self.sim_time = float(self.solver.total_turns)
                    self.playing = False

            for sw in list(self.shockwaves):
                sw.progress += dt * sw.speed
                if sw.progress >= 1.0:
                    self.shockwaves.remove(sw)

            # Render all layers in order.
            self._draw_atmosphere(screen, elapsed)
            self._draw_connections(screen, elapsed)
            node_screen_pos = self._draw_hubs(screen, elapsed)
            drone_states, groups = self._compute_all_drones_render_state(elapsed)
            self._draw_drones(screen, drone_states)

def clamp(val: float, lo: float, hi: float) -> float:
    return max(lo, min(hi, val))


def smoothstep(t: float) -> float:
    t = clamp(t, 0.0, 1.0)
    return t * t * (3.0 - 2.0 * t)


def ease_in_out_sine(t: float) -> float:
    t = clamp(t, 0.0, 1.0)
    return -0.5 * (math.cos(math.pi * t) - 1.0)


def exp_decay(current: float, target: float, speed: float, dt: float) -> float:
    """Exponential interpolation independent of frame rate (no flicker)."""
    factor = 1.0 - math.exp(-speed * dt)
    return current + (target - current) * factor

class GlowCache:
    """Pre-render and cache soft radial glows to maintain 60 FPS."""

    def __init__(self) -> None:
        self._cache: dict[tuple[int, tuple[int, int, int], int], pygame.Surface] = {}

    def get(
        self, radius: int, color: tuple[int, int, int], peak_alpha: int = 80
    ) -> pygame.Surface:
        r_quant = max(4, (radius // 2) * 2)
        c_quant = (color[0] // 4 * 4, color[1] // 4 * 4, color[2] // 4 * 4)
        key = (r_quant, c_quant, peak_alpha)

        if key not in self._cache:
            if len(self._cache) > 512:
                self._cache.clear()
            size = r_quant * 2
            surf = pygame.Surface((size, size), pygame.SRCALPHA)
            layers = 8
            for i in range(layers, 0, -1):
                frac = i / layers
                sub_r = max(1, int(r_quant * frac))
                # Smooth Gaussian falloff toward the edge.
                falloff = math.exp(-3.2 * (frac * frac))
                alpha = int(clamp(peak_alpha * falloff, 0, 255))
                if alpha > 0:
                    pygame.draw.circle(
                        surf, (*c_quant, alpha), (r_quant, r_quant), sub_r
                    )
            self._cache[key] = surf
        return self._cache[key]

    def draw(
        self,
        target: pygame.Surface,
        center: tuple[int, int],
        radius: int,
        color: tuple[int, int, int],
        peak_alpha: int = 80,
    ) -> None:
        if radius <= 2:
            return
        sprite = self.get(radius, color, peak_alpha)
        half = sprite.get_width() // 2
        target.blit(
            sprite,
            (center[0] - half, center[1] - half),
            special_flags=pygame.BLEND_PREMULTIPLIED if False else 0,
        )

class OrganicLabelEngine:
    """
    Calculate optimal label positions using initial topological analysis, then
    apply continuously damped repulsion forces at runtime (no discrete jumps).
    """

    def __init__(self, graph: Graph) -> None:
        self.graph = graph
        self.node_labels: dict[str, SmoothLabelState] = {}
        self.drone_labels: dict[str, SmoothLabelState] = {}
        self._init_topological_node_offsets()

    def _init_topological_node_offsets(self) -> None:
        """
        For each hub, analyze the angles of its connected edges and place its
        label along the bisector of the largest open angle.
        """
        for name, hub in self.graph.nodes_dict.items():
            neighbors = self.graph.get_neighbors(name)
            edge_angles: list[float] = []

            for link in neighbors:
                other_name = link.get_destination(name)
                other = self.graph.nodes_dict.get(other_name)
                if other is not None:
                    # The screen's Y axis is inverted (-dy).
                    dx = float(other.x - hub.x)
                    dy = float(-(other.y - hub.y))
                    if abs(dx) > 1e-4 or abs(dy) > 1e-4:
                        ang = math.atan2(dy, dx) % (2.0 * math.pi)
                        edge_angles.append(ang)

            if not edge_angles:
                best_angle = math.pi * 0.5  # Below by default.
            elif len(edge_angles) == 1:
                best_angle = (edge_angles[0] + math.pi) % (2.0 * math.pi)
            else:
                edge_angles.sort()
                best_gap = -1.0
                best_angle = math.pi * 0.5
                n = len(edge_angles)
                for i in range(n):
                    a1 = edge_angles[i]
                    a2 = (
                        edge_angles[(i + 1) % n]
                        if i + 1 < n
                        else edge_angles[0] + 2.0 * math.pi
                    )
                    gap = a2 - a1
                    mid = (a1 + a2) * 0.5
                    # Slight aesthetic preference for placing labels above or below.
                    vertical_bonus = 0.15 * abs(math.sin(mid))
                    if gap + vertical_bonus > best_gap:
                        best_gap = gap + vertical_bonus
                        best_angle = mid % (2.0 * math.pi)

            dist = 28.0
            rx = math.cos(best_angle) * dist * 1.15
            ry = math.sin(best_angle) * dist
            self.node_labels[name] = SmoothLabelState(
                rest_dx=rx, rest_dy=ry, cur_dx=rx, cur_dy=ry
            )

    def update_node_labels(
        self,
        screen_positions: dict[str, tuple[int, int]],
        node_radius: int,
        dt: float,
    ) -> None:
        """
        Smoothly relax node labels relative to one another using continuous
        forces, without allowing abrupt jumps from one side to the other.
        """
        names = list(self.node_labels.keys())
        base_scale = max(0.85, min(1.45, node_radius / 14.0))

        # 1. Calculate smooth repulsion forces between nearby node labels.
        forces: dict[str, list[float]] = {n: [0.0, 0.0] for n in names}

        for i in range(len(names)):
            n1 = names[i]
            st1 = self.node_labels[n1]
            ax1, ay1 = screen_positions[n1]
            cx1 = ax1 + st1.cur_dx
            cy1 = ay1 + st1.cur_dy

            # Smoothly repel nearby nodes and their labels.
            for j in range(i + 1, len(names)):
                n2 = names[j]
                st2 = self.node_labels[n2]
                ax2, ay2 = screen_positions[n2]

                # Ignore nodes that are far apart on screen.
                if abs(ax1 - ax2) > 220 or abs(ay1 - ay2) > 140:
                    continue

                cx2 = ax2 + st2.cur_dx
                cy2 = ay2 + st2.cur_dy

                # Overlap between the two text boxes.
                min_sep_x = (st1.width + st2.width) * 0.5 + 8.0
                min_sep_y = (st1.height + st2.height) * 0.5 + 6.0
                dx = cx1 - cx2
                dy = cy1 - cy2

                if abs(dx) < min_sep_x and abs(dy) < min_sep_y:
                    pen_x = (min_sep_x - abs(dx)) / min_sep_x
                    pen_y = (min_sep_y - abs(dy)) / min_sep_y
                    strength = pen_x * pen_y * 28.0
                    dir_x = 1.0 if dx >= 0 else -1.0
                    dir_y = 1.0 if dy >= 0 else -1.0
                    forces[n1][0] += dir_x * strength * 0.45
                    forces[n1][1] += dir_y * strength * 0.85
                    forces[n2][0] -= dir_x * strength * 0.45
                    forces[n2][1] -= dir_y * strength * 0.85

        # 2. Integrate with exponential damping toward (rest + force).
        for name in names:
            st = self.node_labels[name]
            fx, fy = forces[name]
            # Limit displacement so labels never drift too far from their hubs.
            fx = clamp(fx, -36.0, 36.0)
            fy = clamp(fy, -28.0, 28.0)

            target_dx = st.rest_dx * base_scale + fx
            target_dy = st.rest_dy * base_scale + fy

            st.cur_dx = exp_decay(st.cur_dx, target_dx, speed=7.5, dt=dt)
            st.cur_dy = exp_decay(st.cur_dy, target_dy, speed=7.5, dt=dt)

    def update_drone_labels(
        self,
        active_drones: list[tuple[str, float, float]],
        node_screen_pos: dict[str, tuple[int, int]],
        drone_radius: int,
        dt: float,
    ) -> None:
        """
        Update drone label positions organically so labels separate smoothly
        when drones travel together.
        """
        rest_y = -(drone_radius + 15.0)
        n = len(active_drones)

        for idx, (did, _, _) in enumerate(active_drones):
            if did not in self.drone_labels:
                # Create a small initial fan based on the drone index.
                fan_x = ((idx % 3) - 1) * 4.0
                self.drone_labels[did] = SmoothLabelState(
                    rest_dx=fan_x,
                    rest_dy=rest_y,
                    cur_dx=fan_x,
                    cur_dy=rest_y,
                )
            else:
                self.drone_labels[did].rest_dy = rest_y

        forces: dict[str, list[float]] = {
            did: [0.0, 0.0] for did, _, _ in active_drones
        }

        # Smoothly repel labels of nearby active drones.
        for i in range(n):
            d1, x1, y1 = active_drones[i]
            st1 = self.drone_labels[d1]
            lx1, ly1 = x1 + st1.cur_dx, y1 + st1.cur_dy

            for j in range(i + 1, n):
                d2, x2, y2 = active_drones[j]
                if abs(x1 - x2) > 130 or abs(y1 - y2) > 90:
                    continue
                st2 = self.drone_labels[d2]
                lx2, ly2 = x2 + st2.cur_dx, y2 + st2.cur_dy

                sep_x = (st1.width + st2.width) * 0.5 + 6.0
                sep_y = (st1.height + st2.height) * 0.5 + 5.0
                dx = lx1 - lx2
                dy = ly1 - ly2

                if abs(dx) < sep_x and abs(dy) < sep_y:
                    # Break exact-position ties deterministically using the ID.
                    if abs(dx) < 0.5 and abs(dy) < 0.5:
                        angle = (i - j) * 1.1
                        dx, dy = math.cos(angle), math.sin(angle)
                    overlap_x = (sep_x - abs(dx)) / sep_x
                    overlap_y = (sep_y - abs(dy)) / sep_y
                    push = overlap_x * overlap_y * 24.0
                    sx = 1.0 if dx >= 0 else -1.0
                    sy = 1.0 if dy >= 0 else -1.0
                    forces[d1][0] += sx * push * 0.7
                    forces[d1][1] += sy * push * 0.7
                    forces[d2][0] -= sx * push * 0.7
                    forces[d2][1] -= sy * push * 0.7

            # Smoothly repel nearby visible node labels.
            for hname, (hx, hy) in node_screen_pos.items():
                if abs(lx1 - hx) > 140 or abs(ly1 - hy) > 90:
                    continue
                nst = self.node_labels.get(hname)
                if nst is None:
                    continue
                nlx, nly = hx + nst.cur_dx, hy + nst.cur_dy
                sep_x = (st1.width + nst.width) * 0.5 + 6.0
                sep_y = (st1.height + nst.height) * 0.5 + 5.0
                dx = lx1 - nlx
                dy = ly1 - nly
                if abs(dx) < sep_x and abs(dy) < sep_y:
                    push = ((sep_y - abs(dy)) / sep_y) * 18.0
                    sy = -1.0 if dy <= 0 else 1.0
                    forces[d1][1] += sy * push

        for did, _, _ in active_drones:
            st = self.drone_labels[did]
            fx = clamp(forces[did][0], -34.0, 34.0)
            fy = clamp(forces[did][1], -30.0, 22.0)
            target_dx = st.rest_dx + fx
            target_dy = st.rest_dy + fy
            st.cur_dx = exp_decay(st.cur_dx, target_dx, speed=9.0, dt=dt)
            st.cur_dy = exp_decay(st.cur_dy, target_dy, speed=9.0, dt=dt)


class SmoothCamera2D:
    """2D camera with exponential interpolation for smooth zooming and panning."""

    def __init__(self, width: int, height: int) -> None:
        self.width = width
        self.height = height
        self.offset_x = 0.0
        self.offset_y = 0.0
        self.zoom = 90.0

        self.target_offset_x = 0.0
        self.target_offset_y = 0.0
        self.target_zoom = 90.0

    def fit_graph(self, graph: Graph, instant: bool = False) -> None:
        if not graph.nodes_dict:
            return
        xs = [n.x for n in graph.nodes_dict.values()]
        ys = [n.y for n in graph.nodes_dict.values()]
        min_x, max_x = min(xs), max(xs)
        min_y, max_y = min(ys), max(ys)

        span_x = max(1.0, float(max_x - min_x))
        span_y = max(1.0, float(max_y - min_y))

        margin_x = 160
        margin_y = 145
        usable_w = max(240, self.width - 2 * margin_x)
        usable_h = max(220, self.height - 2 * margin_y)

        z = min(usable_w / span_x, usable_h / span_y)
        z = clamp(z, 18.0, 175.0)

        cx = (min_x + max_x) * 0.5
        cy = (min_y + max_y) * 0.5

        self.target_zoom = z
        self.target_offset_x = (self.width * 0.5) - cx * z
        self.target_offset_y = (self.height * 0.5) + cy * z

        if instant:
            self.zoom = self.target_zoom
            self.offset_x = self.target_offset_x
            self.offset_y = self.target_offset_y

    def update(self, dt: float) -> None:
        self.zoom = exp_decay(self.zoom, self.target_zoom, 12.0, dt)
        self.offset_x = exp_decay(self.offset_x, self.target_offset_x, 12.0, dt)
        self.offset_y = exp_decay(self.offset_y, self.target_offset_y, 12.0, dt)

    def pan(self, dx: float, dy: float) -> None:
        self.target_offset_x += dx
        self.target_offset_y += dy
        self.offset_x += dx
        self.offset_y += dy

    def zoom_at(self, mouse_pos: tuple[int, int], factor: float) -> None:
        mx, my = mouse_pos
        wx = (mx - self.target_offset_x) / self.target_zoom
        wy = (self.target_offset_y - my) / self.target_zoom

        new_z = clamp(self.target_zoom * factor, 10.0, 380.0)
        self.target_zoom = new_z
        self.target_offset_x = mx - wx * new_z
        self.target_offset_y = my + wy * new_z

    def to_screen(self, x: float, y: float) -> tuple[int, int]:
        return int(round(self.offset_x + x * self.zoom)), int(
            round(self.offset_y - y * self.zoom)
        )

    def to_screen_float(self, x: float, y: float) -> tuple[float, float]:
        return self.offset_x + x * self.zoom, self.offset_y - y * self.zoom
